OOP Basics
class Animal:
    kind = "Mammal"           # class attr

    def __init__(self, name):
        self.name = name      # instance attr

    def speak(self):          # method
        return f"{self.name} speaks"

    @classmethod
    def info(cls): return cls.kind

    @staticmethod
    def breathe(): return "breathing"

    def __str__(self):
        return f"Animal({self.name})"

a = Animal("Dog")
print(a.speak())
print(Animal.info())

class Shape(ABC):

Sorting Algorithms
# Bubble Sort O(n²)
def bubble(arr):
    for i in range(len(arr)):
        for j in range(len(arr)-i-1):
            if arr[j] > arr[j+1]:
                arr[j],arr[j+1] = arr[j+1],arr[j]

# Merge Sort O(n log n)
def merge_sort(arr):
    if len(arr)<=1: return arr
    mid = len(arr)//2
    L = merge_sort(arr[:mid])
    R = merge_sort(arr[mid:])
    return merge(L, R)

# Quick Sort O(n log n) avg
def quick(arr):
    if len(arr)<=1: return arr
    p = arr[len(arr)//2]
    return quick([x for x in arr if x<p]) + \
           [x for x in arr if x==p] + \
           quick([x for x in arr if x>p])
Searching
# Linear Search O(n)
def linear(arr, t):
    for i,v in enumerate(arr):
        if v == t: return i
    return -1

# Binary Search O(log n) — sorted only
def binary(arr, t):
    lo, hi = 0, len(arr)-1
    while lo <= hi:
        mid = (lo+hi)//2
        if arr[mid] == t: return mid
        elif arr[mid] < t: lo = mid+1
        else: hi = mid-1
    return -1
